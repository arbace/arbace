package libm

import "math"

// Hypot is musl's hypot (src/math/hypot.c, musl 1.2.6): sqrt(x*x + y*y) without undue overflow or
// underflow, the squares split exactly (Dekker) so the sum is rounded once. Ported for pyi (complex
// abs and pow, math.hypot; queue 186, PY5e part 2b). Every product is converted to float64 on its own:
// Go may fuse a*b+c, which musl's C does not (FLT_EVAL_METHOD 0, no contraction).
func Hypot(x, y float64) float64 {
	ux := math.Float64bits(x) &^ (1 << 63)
	uy := math.Float64bits(y) &^ (1 << 63)
	if ux < uy {
		ux, uy = uy, ux
	}
	ex := int(ux >> 52)
	ey := int(uy >> 52)
	x = math.Float64frombits(ux)
	y = math.Float64frombits(uy)
	if ey == 0x7ff {
		return y
	}
	if ex == 0x7ff || uy == 0 {
		return x
	}
	if ex-ey > 64 {
		return x + y
	}
	z := 1.0
	if ex > 0x3ff+510 {
		z = 0x1p700
		x *= 0x1p-700
		y *= 0x1p-700
	} else if ey < 0x3ff-450 {
		z = 0x1p-700
		x *= 0x1p700
		y *= 0x1p700
	}
	hx, lx := hypotSq(x)
	hy, ly := hypotSq(y)
	return z * math.Sqrt(ly+lx+hy+hx)
}

// hypotSq is musl's sq: x*x as hi + lo, exactly.
func hypotSq(x float64) (hi, lo float64) {
	const split = 0x1p27 + 1
	xc := float64(x * split)
	xh := x - xc + xc
	xl := x - xh
	hi = float64(x * x)
	lo = float64(xh*xh) - hi + float64(2*xh*xl) + float64(xl*xl)
	return hi, lo
}
