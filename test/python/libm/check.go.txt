package main

import (
	"arbace/libm"
	"math"
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

// reads NAME A B R lines (hex), recomputes with the port, prints per-function counts and the first mismatches
func main() {
	sc := bufio.NewScanner(os.Stdin)
	sc.Buffer(make([]byte, 1<<20), 1<<20)
	type c struct{ n, bad int }
	cnt := map[string]*c{}
	var order []string
	shown := 0
	for sc.Scan() {
		f := strings.Fields(sc.Text())
		a, _ := strconv.ParseUint(f[1], 16, 64)
		b, _ := strconv.ParseUint(f[2], 16, 64)
		r, _ := strconv.ParseUint(f[3], 16, 64)
		got, ok := call(f[0], a, b)
		if !ok {
			panic(f[0])
		}
		k := cnt[f[0]]
		if k == nil {
			k = &c{}
			cnt[f[0]] = k
			order = append(order, f[0])
		}
		k.n++
		if got != r {
			k.bad++
			if shown < 15 {
				fmt.Printf("mismatch %s %s %s: musl %x port %x\n", f[0], f[1], f[2], r, got)
				shown++
			}
		}
	}
	tot := 0
	for _, n := range order {
		fmt.Println(n, cnt[n].n, cnt[n].bad)
		tot += cnt[n].bad
	}
	fmt.Println("TOTAL mismatches", tot)
}

func call(name string, a, b uint64) (uint64, bool) {
	if name == "hypot" {
		return math.Float64bits(libm.Hypot(math.Float64frombits(a), math.Float64frombits(b))), true
	}
	return libm.Call(name, a, b)
}
