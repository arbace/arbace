// hy: random and edge-case inputs of the ported libm functions, with the port's result bits, one line each
package main

import (
	"arbace/libm"
	"fmt"
	"math"
	"math/rand"
	"os"
	"strconv"
)

func main() {
	n, _ := strconv.Atoi(os.Args[1])
	r := rand.New(rand.NewSource(1))
	edges := []float64{0, math.Copysign(0, -1), 1, -1, math.Inf(1), math.Inf(-1), math.NaN(), math.MaxFloat64, math.SmallestNonzeroFloat64, 0x1p-1022, 0x1p1023, 3, 4, 1e-300, 1e300}
	emit := func(x, y float64) {
		fmt.Printf("%016x %016x %016x\n", math.Float64bits(x), math.Float64bits(y), math.Float64bits(libm.Hypot(x, y)))
	}
	for _, a := range edges {
		for _, b := range edges {
			emit(a, b)
		}
	}
	for i := 0; i < n; i++ {
		var x, y float64
		switch i % 3 {
		case 0: // any bit patterns
			x, y = math.Float64frombits(r.Uint64()), math.Float64frombits(r.Uint64())
		case 1: // ordinary magnitudes
			x, y = (r.Float64()-0.5)*math.Pow(10, float64(r.Intn(40)-20)), (r.Float64()-0.5)*math.Pow(10, float64(r.Intn(40)-20))
		default: // close exponents
			x = r.NormFloat64() * 1e10
			y = x * (1 + r.Float64())
		}
		emit(x, y)
	}
}
